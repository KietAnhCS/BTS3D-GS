    \footnotesize
    \begin{itemize}

\begin{frame}{Sơ đồ quyết định densify đầy đủ}
\begin{columns}[T]
\begin{column}{0.48\linewidth}
\footnotesize
\begin{itemize}
  \item Áp dụng cho mỗi Gaussian $i$, mỗi 100 (hoặc 500) vòng lặp, khi $500 < t < 15000$.
  \item \textbf{Bước 1 — kích thước chọn nhánh:} $\max s_i \le \delta \cdot \text{extent}\;?$
  \item \textbf{Nhánh NHỎ:} nếu $\bar g_i \ge \tau_{\text{grad}} = 2\times10^{-4}$ \textbf{và} $\texttt{accum\_view\_count}_i > 0.5$ $\Rightarrow$ \textbf{CLONE} ($1 \to 2$, giữ bản gốc).
  \item \textbf{Nhánh TO:} nếu $\bar g_{\text{abs},i} \ge \tau_{\text{abs}} = 2\times10^{-4}$ \textbf{và} $\texttt{accum\_view\_count}_i > 0.5$ $\Rightarrow$ \textbf{SPLIT} ($N = k_xk_yk_z$ con, xoá gốc).
  \item Nếu Importance không thoả: 3DGS gốc vẫn densify, cơ chế này thì không (SADGS bổ sung thêm so khớp cấu trúc texture đa tỉ lệ, xem BOOK/17).
  \item Nguồn: \texttt{densify\_and\_prune\_structgs}, \texttt{gaussian\_model.py:957-1053}.
\end{itemize}
\end{column}
